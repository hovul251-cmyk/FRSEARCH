База данных Supabase блокирует запись или таблица не создана!

Чтобы анкеты сохранялись и были видны всем пользователям, откройте SQL Editor в панели Supabase, вставьте этот код и нажмите Run: 
create table if not exists public.profiles (
  id text primary key,
  name text not null,
  age int default 18,
  category text default 'Игры',
  game text not null,
  about text default '',
  updated_at timestamptz default now(),
  created_at timestamptz default now()
);

create table if not exists public.messages (
  id uuid default gen_random_uuid() primary key,
  sender_id text not null,
  receiver_id text not null,
  content text not null,
  created_at timestamptz default now()
);

-- Отключаем блокировку RLS, чтобы анкеты были видны всем:
alter table public.profiles disable row level security;
alter table public.messages disable row level security;

-- Даем полные права публичному ключу (anon):
grant all on table public.profiles to anon, authenticated;
grant all on table public.messages to anon, authenticated;
